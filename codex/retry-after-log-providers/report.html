func updateAtomicMin(a atomicInt64, candidate int64) {
        for {
                current := a.Load()
                if current == -1 || current > candidate {

                break
        }

func (cs *CertStream) LogInfo(msg string, args ...any) {

func (cs *CertStream) LogError(err error, msg string, args ...any) error {
        if err != nil && cs.Logger != nil {
                if !errors.Is(err, context.Canceled) && !errors.Is(err, context.DeadlineExceeded) {
                        if unwrapper, ok := err.(interface{ Unwrap() []error }); ok {
                                for _, e := range unwrapper.Unwrap() {
                                        cs.Logger.Error("certstream: "+msg, append(args, "err", e)...)
                                }
                        } else {
                                cs.Logger.Error("certstream: "+msg, append(args, "err", err)...)
                        }
                }
        }
        return err
}

func (cs *CertStream) Operators() (operators []*LogOperator) {
        cs.mu.Lock()

        return

func (cs *CertStream) close() {
        cs.mu.Lock()
        seCh := cs.sendEntryCh
        cs.sendEntryCh = nil
        db := cs.db
        cs.db = nil
        headLog := cs.headLogFile
        cs.headLogFile = nil
        tailLog := cs.tailLogFile
        cs.tailLogFile = nil
        cs.mu.Unlock()
        if seCh != nil {
                // drain
                ok := true
                for ok {
                        select {
                        case _, ok = <-seCh:
                        default:
                                ok = false
                        }
                }
                close(seCh)
        }
        if db != nil {
                db.logGaps()
                db.Close()
        }
        if headLog != nil {
                _ = headLog.Close()
        }
        if tailLog != nil {
                _ = tailLog.Close()
        }
}

func (cs *CertStream) run(ctx context.Context, pwg *sync.WaitGroup) {
        var streamWG sync.WaitGroup
        var dbWG sync.WaitGroup
        ticker := time.NewTicker(time.Hour * 24)

        defer func() {
                streamWG.Wait()
                dbWG.Wait()
                cs.close()
                pwg.Done()
                ticker.Stop()
        }()

        if db := cs.DB(); db != nil {
                dbWG.Add(1)
                go db.runWorkers(ctx, &dbWG)
        }

        streamWG.Add(1)
        _ = cs.LogError(cs.updateStreams(ctx, &streamWG), "CertStream:run@1")

        if db := cs.DB(); db != nil {
                dbWG.Add(3)
                go db.estimator(ctx, &dbWG)
                go db.reportGaps(ctx, &dbWG)
                go func() {
                        defer dbWG.Done()
                        db.CleanCertificates(ctx)
                }()
        }

        for {
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                        streamWG.Add(1)
                        _ = cs.LogError(cs.updateStreams(ctx, &streamWG), "CertStream:run@2")
                }
        }
}

// Start starts CT log discovery and streaming until ctx is canceled.
//
// It adds background goroutines to wg and returns after setup.
func Start(ctx context.Context, wg *sync.WaitGroup, cfg *Config) (cs *CertStream, err error) {
        tailDialer := cfg.TailDialer
        if tailDialer == nil {
                tailDialer = cfg.HeadDialer
        }
        tphead := DefaultTransport.Clone()
        tphead.DialContext = cfg.HeadDialer.DialContext
        headTransport := http.RoundTripper(tphead)
        var headLogFile *os.File
        if cfg.HeadLog != "" {
                headLogFile, err = os.OpenFile(cfg.HeadLog, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
                if err == nil {
                        headTransport = newHeadLogTransport(tphead, headLogFile)
                } else {
                        err = errHeadLogOpen{err: err}
                }
        }
        tptail := DefaultTransport.Clone()
        tptail.DialContext = tailDialer.DialContext
        tailTransport := http.RoundTripper(tptail)
        var tailLogFile *os.File
        if err == nil {
                if cfg.TailLog != "" {
                        tailLogFile, err = os.OpenFile(cfg.TailLog, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
                        if err == nil {
                                tailTransport = newTailLogTransport(tptail, tailLogFile)
                        } else {
                                err = errTailLogOpen{err: err}
                        }
                }
        }

        if cfg.DataDir != "" {
                cfg.DataDir = strings.TrimSpace(os.ExpandEnv(cfg.DataDir))
        }

        if err == nil && cfg.DataDir != "" {
                if cfg.CacheMaxAge > 0 {
                        err = os.MkdirAll(getCacheDir(cfg.DataDir, ""), 0o755)
                }
        }

        if err == nil {
                cs = &CertStream{
                        Config: *cfg,
                        HeadClient: &http.Client{
                                Timeout:   30 * time.Second,
                                Transport: httpCallCounter{headTransport},
                        },
                        TailClient: &http.Client{
                                Timeout:   30 * time.Second,
                                Transport: httpCallCounter{tailTransport},
                        },
                        headLogFile: headLogFile,
                        tailLogFile: tailLogFile,
                        operators:   map[string]*LogOperator{},
                }

                var db *PgDB
                if db, err = NewPgDB(ctx, cs); err == nil {
                        cs.mu.Lock()
                        cs.db = db
                        cs.sendEntryCh = make(chan *LogEntry, 1024*8)
                        cs.C = cs.sendEntryCh
                        cs.mu.Unlock()
                        if cs.DataDir != "" && cs.CacheMaxAge > 0 {
                                wg.Add(1)
                                go cs.runCachePruner(ctx, wg, getCacheDir(cs.DataDir, ""), cs.CacheMaxAge)
                        }
                        wg.Add(1)
                        go cs.run(ctx, wg)
                }
        }
        if err != nil && tailLogFile != nil {
                _ = tailLogFile.Close()
                if cs != nil {
                        cs.mu.Lock()
                        if cs.tailLogFile == tailLogFile {
                                cs.tailLogFile = nil
                        }
                        cs.mu.Unlock()
                }
        }
        if err != nil && headLogFile != nil {
                _ = headLogFile.Close()
                if cs != nil {
                        cs.mu.Lock()
                        if cs.headLogFile == headLogFile {
                                cs.headLogFile = nil
                        }
                        cs.mu.Unlock()
                }
        }

        return
}

                                }

                        }
                }
        }
        return

type httpCallCounter struct {
        next http.RoundTripper
}

func (sct httpCallCounter) RoundTrip(req *http.Request) (*http.Response, error) {
        mapkey := req.URL.Scheme + "://" + req.URL.Host + req.URL.Path
        for _, cutoff := range []string{"/ct/", "/tile/", "/checkpoint"} {
                if idx := strings.Index(mapkey, cutoff); idx != -1 {
                        mapkey = mapkey[:idx+1]
                        break
                }
        }
        httpCallsMu.Lock()
        httpCallsMap[mapkey]++
        httpCallsMu.Unlock()
        return sct.next.RoundTrip(req)
}

func GetHTTPCallsMap() (m map[string]int64) {
        httpCallsMu.Lock()
        m = maps.Clone(httpCallsMap)
        httpCallsMu.Unlock()
        return
}

func GetHTTPCalls(s string) (n int64) {
        httpCallsMu.Lock()
        n = httpCallsMap[s]
        httpCallsMu.Unlock()
        return
}

func NewJSONCertificate(cert *Certificate) (jsoncert *JsonCertificate) {
        jsoncert = &JsonCertificate{
                PreCert:    cert.PreCert,
                Signature:  cert.Signature,
                CommonName: cert.GetCommonName(),
                NotBefore:  cert.NotBefore,
                NotAfter:   cert.NotAfter,
        }
        jsoncert.Issuer.fill(&cert.Issuer)
        jsoncert.Subject.fill(&cert.Subject)

        for _, dnsname := range cert.DNSNames {
                dnsname = strings.ToLower(dnsname)
                if uniname, err := idna.ToUnicode(dnsname); err == nil && uniname != dnsname {
                        jsoncert.DNSNames = append(jsoncert.DNSNames, uniname)
                } else {
                        jsoncert.DNSNames = append(jsoncert.DNSNames, dnsname)
                }
        }
        sort.Strings(jsoncert.DNSNames)

        for _, ip := range cert.IPAddresses {
                jsoncert.IPAddresses = append(jsoncert.IPAddresses, ip.String())
        }
        sort.Strings(jsoncert.IPAddresses)

        for _, email := range cert.EmailAddresses {
                if m, e := mail.ParseAddress(email); e == nil {
                        email = m.Address
                }
                jsoncert.EmailAddresses = append(jsoncert.EmailAddresses, email)
        }
        sort.Strings(jsoncert.EmailAddresses)

        for _, u := range cert.URIs {
                jsoncert.URIs = append(jsoncert.URIs, u.String())
        }
        sort.Strings(jsoncert.URIs)
        return
}

func (le *LogEntry) appendJSON(b []byte) []byte {
        if cert := le.Cert(); cert != nil {
                logindex := le.Index()
                var dnsnames []string
                for _, dnsname := range cert.DNSNames {
                        dnsname = strings.ToLower(dnsname)
                        if uniname, err := idna.ToUnicode(dnsname); err == nil && uniname != dnsname {
                                ok := true
                                for _, r := range uniname {
                                        ok = ok && strconv.IsPrint(r)
                                }
                                if ok {
                                        dnsname = uniname
                                }
                        }
                        dnsnames = append(dnsnames, dnsname)
                }

                var ipaddrs []string
                for _, ip := range cert.IPAddresses {
                        ipaddrs = append(ipaddrs, ip.String())
                }

                var emails []string
                for _, email := range cert.EmailAddresses {
                        emails = append(emails, strings.ReplaceAll(email, " ", "_"))
                }

                var uris []string
                for _, uri := range cert.URIs {
                        uris = append(uris, strings.ReplaceAll(uri.String(), " ", "%20"))
                }

                b = append(b, `{`...)
                b = append(b, `"iss_org":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Issuer.Organization, ","))
                b = append(b, `,"iss_prov":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Issuer.Province, ","))
                b = append(b, `,"iss_country":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Issuer.Country, ","))
                b = append(b, `,"sub_org":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Subject.Organization, ","))
                b = append(b, `,"sub_prov":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Subject.Province, ","))
                b = append(b, `,"sub_country":`...)
                b = strconv.AppendQuote(b, strings.Join(cert.Subject.Country, ","))
                b = append(b, `,"notbefore":"`...)
                b = append(b, cert.NotBefore.UTC().Format(time.RFC3339)...)
                b = append(b, `","notafter":"`...)
                b = append(b, cert.NotAfter.UTC().Format(time.RFC3339)...)
                b = append(b, `","commonname":`...)
                b = strconv.AppendQuote(b, cert.GetCommonName())
                b = append(b, `,"sha256_hex":"`...)
                b = append(b, hex.EncodeToString(cert.Signature)...)
                b = append(b, `","precert":`...)
                b = strconv.AppendBool(b, cert.PreCert)
                b = append(b, `,"seen":"`...)
                b = append(b, cert.Seen.UTC().Format(time.RFC3339)...)
                b = append(b, `","stream":`...)
                b = strconv.AppendInt(b, int64(le.LogStream.Id), 10)
                b = append(b, `,"logindex":`...)
                b = strconv.AppendInt(b, logindex, 10)
                b = append(b, `,"dnsnames":`...)
                b = strconv.AppendQuote(b, strings.Join(dnsnames, " "))
                b = append(b, `,"ipaddrs":`...)
                b = strconv.AppendQuote(b, strings.Join(ipaddrs, " "))
                b = append(b, `,"emails":`...)
                b = strconv.AppendQuote(b, strings.Join(emails, " "))
                b = append(b, `,"uris":`...)
                b = strconv.AppendQuote(b, strings.Join(uris, " "))
                b = append(b, `}`...)
        } else {
                b = append(b, `{}`...)
        }
        return b
}

func (le *LogEntry) String() (s string) {
        var b []byte
        b = append(b, "LogEntry{"...)
        if le != nil {
                if le.LogStream != nil {
                        if le.operator != nil {
                                b = strconv.AppendQuote(b, le.operator.Name)
                                b = append(b, ", "...)
                        }
                        if url := le.URL(); url != "" {
                                b = strconv.AppendQuote(b, url)
                                b = append(b, ", "...)
                        }
                }
                b = strconv.AppendInt(b, le.Index(), 10)
                if le.Err != nil {
                        b = append(b, ", "...)
                        b = strconv.AppendQuote(b, le.Err.Error())
                }
        }
        b = append(b, '}')
        return string(b)
}

// Cert returns the Certificate given a LogEntry or nil.
func (le *LogEntry) Cert() (crt *Certificate) {
        if le != nil && le.Certificate != nil {
                crt = &Certificate{
                        PreCert:     le.PreCert,
                        Certificate: le.Certificate,
                        Signature:   le.Signature,
                        Seen:        le.Seen,
                }
                if le.Seen.IsZero() {
                        crt.Seen = time.Now().UTC()
                }
        }
        return
}

// Index returns the log index or -1 if none is available.
func (le *LogEntry) Index() (index int64) {
        index = -1
        if le != nil {
                if le.LogIndex != 0 || le.Certificate != nil || le.PreCert || le.Err != nil {
                        index = le.LogIndex
                }
        }
        return
}

func (ls *LogStream) run(ctx context.Context, wg *sync.WaitGroup) {
        var end int64
        var err error
        var wg2 sync.WaitGroup
        defer func() {
                if ls.cancel != nil {
                        ls.cancel()
                }
                if !errors.Is(err, context.Canceled) {
                        ls.addError(ls, err)
                }
                wg2.Wait()
                ls.removeStream(ls)
                if e, ok := err.(errLogIdle); ok {
                        ls.LogInfo("stream stopped", "url", ls.URL(), "stream", ls.Id, "idle-since", e.Since)
                } else {
                        _ = ls.LogError(err, "stream stopped", "url", ls.URL(), "stream", ls.Id)
                }
                wg.Done()
                if ls.done != nil {
                        close(ls.done)
                }
        }()

        end, err = ls.newLastIndex(ctx)
        if seen := ls.getEndSeen(ctx, end); !seen.IsZero() {
                if time.Since(seen) > IdleCloseTime {
                        err = errLogIdle{Since: seen}
                        return
                }
        }

        start := end
        if cdb := ls.DB(); cdb != nil {
                if ls.TailDialer != nil {
                        backfillCtx, cancelBackfill := context.WithCancel(ctx)
                        defer cancelBackfill()
                        wg2.Add(1)
                        go cdb.backfillStream(backfillCtx, ls, &wg2)
                }
        }

        for err == nil {
                if start <= end {
                        startBefore := start
                        start, _ = ls.getEntries(ctx, start, end, false, ls.sendEntry, nil)
                        if end-startBefore <= LogBatchSize/2 {
                                _ = sleep(ctx, time.Second*time.Duration(10+rand.IntN(10) /*#nosec G404*/))
                        }
                }
                end, err = ls.newLastIndex(ctx)
        }
}

func (ls *LogStream) newLastIndex(ctx context.Context) (lastIndex int64, err error) {
        started := time.Now()
        var newIndex int64
        lastIndex = ls.LastIndex.Load()
        for err == nil {
                err = ls.backoff.Retry(ctx, func() (callErr error) {
                        var errFrom string
                        if ls.isTiled() {
                                errFrom = "Checkpoint"
                                if ls.headTile != nil {
                                        var checkpoint torchwood.Checkpoint
                                        checkpoint, _, callErr = ls.headTile.Checkpoint(ctx)
                                        if callErr == nil {
                                                newIndex = checkpoint.N - 1
                                        }
                                } else {
                                        callErr = ErrSunlightClientMissing
                                }
                        } else {
                                errFrom = "GetSTH"
                                var sth *ct.SignedTreeHead
                                sth, callErr = ls.headClient.GetSTH(ctx)
                                if callErr == nil {
                                        newIndex = int64(sth.TreeSize) - 1 //#nosec G115
                                }
                        }
                        if callErr != nil {
                                if ls.handleStreamError(callErr, errFrom) {
                                        // keep callErr as-is
                                } else {
                                        callErr = wrapLogStreamRetryable(callErr)
                                }
                        }
                        return
                })
                if err == nil {
                        if lastIndex < newIndex && (lastIndex+LogBatchSize < newIndex || time.Since(started) > time.Minute) {
                                lastIndex = newIndex
                                ls.LastIndex.Store(lastIndex)
                                return
                        }
                        err = sleep(ctx, time.Millisecond*time.Duration(8000+rand.IntN(4000)))
                }
        }
        return
}

func (ls *LogStream) seeIndex(logindex int64) {
        if logindex >= 0 {
                updateAtomicMin(&ls.MinIndex, logindex)
                updateAtomicMax(&ls.MaxIndex, logindex)
        }
}

func (ls *LogStream) makeLogEntry(logindex int64, entry ct.LeafEntry, historical bool) *LogEntry {
        ctrle, leaferr := ct.RawLogEntryFromLeaf(logindex, &entry)
        var ctle *ct.LogEntry
        if leaferr == nil {
                ctle, leaferr = ctrle.ToLogEntry()
        }
        le := &LogEntry{
                LogStream:  ls,
                Err:        leaferr,
                LogIndex:   logindex,
                Historical: historical,
        }
        if ctle != nil {
                if ctle.X509Cert != nil {
                        le.Certificate = ctle.X509Cert
                } else if ctle.Precert != nil {
                        le.PreCert = true
                        le.Certificate = ctle.Precert.TBSCertificate
                }
        }
        if ctrle != nil {
                if len(ctrle.Cert.Data) > 0 {
                        shasig := sha256.Sum256(ctrle.Cert.Data)
                        le.Signature = shasig[:]
                }
                if tse := ctrle.Leaf.TimestampedEntry; tse != nil {
                        ts := int64(tse.Timestamp) //#nosec G115
                        le.Seen = time.UnixMilli(ts).UTC()
                }
        }
        if len(le.Signature) == 0 && le.Certificate != nil {
                if raw := le.Certificate.Raw; len(raw) > 0 {
                        shasig := sha256.Sum256(raw)
                        le.Signature = shasig[:]
                } else if raw := le.Certificate.RawTBSCertificate; len(raw) > 0 {
                        shasig := sha256.Sum256(raw)
                        le.Signature = shasig[:]
                }
        }
        return le
}

func (ls *LogStream) sendEntry(ctx context.Context, now time.Time, le *LogEntry) (wanted bool) {
        if le != nil {
                if cert := le.Cert(); cert != nil {
                        ls.seeIndex(le.LogIndex)
                        wanted = now.Before(cert.NotAfter) || now.Sub(cert.Seen) < time.Hour*24*time.Duration(ls.PgMaxAge)
                        if ctx.Err() == nil {
                                ls.Count.Add(1)
                                ls.LogOperator.Count.Add(1)
                                if db := ls.DB(); db != nil {
                                        db.sendToBatcher(ctx, le)
                                } else {
                                        select {
                                        case <-ctx.Done():
                                        case ls.getSendEntryCh() <- le:
                                        }
                                }
                        }
                } else {
                        wanted = true
                        err := le.Err
                        if err == nil {
                                err = os.ErrInvalid
                        }
                        _ = ls.LogError(err, "sendEntry", "url", ls.URL(), "stream", ls.Id, "logindex", le.LogIndex)
                }
        }
        return
}

func (ls *LogStream) handleStreamError(err error, from string) (fatal bool) {
        errTxt := err.Error()
        underlyingErr := err
        var rspErr jsonclient.RspError
        // jsonclient.RspError keeps its cause in Err without unwrapping it.
        if errors.As(err, &rspErr) {
                underlyingErr = rspErr.Err
        }
        if errors.Is(err, context.Canceled) || errors.Is(underlyingErr, context.Canceled) || errors.Is(err, os.ErrInvalid) || errors.Is(err, io.ErrNoProgress) || strings.Contains(errTxt, "context canceled") {
                fatal = true
        } else if errors.Is(err, context.DeadlineExceeded) || errors.Is(underlyingErr, context.DeadlineExceeded) || strings.Contains(errTxt, "deadline exceeded") {
                fatal = false
        } else {
                statusCode := ls.statusCodeFromError(err)
                switch statusCode {
                default:
                        if ls.addError(ls, wrapErr(err, from)) >= MaxErrors {
                                fatal = true
                        }
                        fallthrough
                case 530: // https://developers.cloudflare.com/support/troubleshooting/http-status-codes/cloudflare-5xx-errors/error-530/
                        fallthrough
                case http.StatusTooManyRequests, http.StatusGatewayTimeout, http.StatusNotFound, http.StatusBadRequest, http.StatusConflict:
                        if ls.LogToggle.Load() && ls.Logger != nil {
                                ls.Logger.Error(from, "url", ls.URL(), "error", err)
                        }
                        ls.addStatus(statusCode)
                }
        }
        return
}

func (ls *LogStream) statusCodeFromError(err error) (code int) {
        if err != nil {
                if rspErr, isRspErr := err.(jsonclient.RspError); isRspErr {
                        if code = rspErr.StatusCode; code >= 500 {
                                // https://developers.cloudflare.com/support/troubleshooting/http-status-codes/cloudflare-5xx-errors/
                                if after, found := bytes.CutPrefix(bytes.TrimSpace(rspErr.Body), []byte("error code:")); found {
                                        if n, err := strconv.Atoi(string(bytes.TrimSpace(after))); err == nil {
                                                if n >= 1000 {
                                                        code = n
                                                }
                                        }
                                }
                                if code < 1000 {
                                        ls.LogInfo("code500+", "url", ls.URL(), "stream", ls.Id, "code", code, "body", string(rspErr.Body))
                                }
                        }
                } else {
                        msg := err.Error()
                        idx := strings.LastIndex(msg, "status code ")
                        if idx >= 0 {
                                start := idx + len("status code ")
                                end := start
                                for end < len(msg) {
                                        if msg[end] < '0' || msg[end] > '9' {
                                                break
                                        }
                                        end++
                                }
                                if end > start {
                                        code, _ = strconv.Atoi(msg[start:end])
                                }
                        }
                }
        }
        return
}

func (ls *LogStream) getEntries(ctx context.Context, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (next int64, wanted bool) {
        next = start
        if start <= end {
                if ls.isTiled() {
                        next, wanted = ls.getTileEntries(ctx, start, end, historical, handleFn, gapcounter)
                } else {
                        var fn func(ctx context.Context, client rawEntriesClient, start int64, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (wanted bool, next int64, err error)
                        client := ls.headClient
                        fn = ls.getRawEntriesParallel
                        if historical {
                                fn = ls.getRawEntries
                                if ls.tailClient != nil {
                                        client = ls.tailClient
                                }
                        }
                        wanted, next, _ = fn(ctx, client, start, end, historical, handleFn, gapcounter)
                }
        }
        return
}

var ErrLogStreamRetryable = errors.New("logstream retryable")

// ErrSTHDiffTooLow indicates a checkpoint does not contain the requested entry.
var ErrSTHDiffTooLow = errors.New("STH diff too low")

type logStreamRetryableError struct {
        err error
}

func (e logStreamRetryableError) Error() string {
        return e.err.Error()
}

func (e logStreamRetryableError) Unwrap() error {
        return e.err
}

func (e logStreamRetryableError) Is(target error) bool {
        return target == ErrLogStreamRetryable
}

func wrapLogStreamRetryable(err error) (wrapped error) {
        if err != nil {
                wrapped = logStreamRetryableError{err: err}
        } else {
                wrapped = ErrLogStreamRetryable
        }
        return
}

type logStreamBackoff struct {
        mu      sync.Mutex
        min     time.Duration
        max     time.Duration
        factor  float64
        jitter  bool
        count   int64
        current time.Duration
        next    time.Time
        nowFn   func() time.Time
        sleepFn func(context.Context, time.Duration) (err error)
}

func newLogStreamBackoff(min, max time.Duration, factor float64, jitter bool) (bo *logStreamBackoff) {
        bo = &logStreamBackoff{
                min:     min,
                max:     max,
                factor:  factor,
                jitter:  jitter,
                nowFn:   time.Now,
                sleepFn: sleep,
        }
        return
}

func (b *logStreamBackoff) Count() (n int64) {
        if b != nil {
                b.mu.Lock()
                n = b.count
                b.mu.Unlock()
        }
        return
}

func (b *logStreamBackoff) Retry(ctx context.Context, f func() error) (err error) {
        for err == nil {
                if err = b.wait(ctx); err == nil {
                        if err = f(); err == nil {
                                b.success()
                                break
                        }
                        if errors.Is(err, ErrLogStreamRetryable) {
                                b.backoff()
                                err = nil
                        }
                }
        }
        return
}

func (b *logStreamBackoff) wait(ctx context.Context) (err error) {
        if b != nil {
                if err = ctx.Err(); err == nil {
                        b.mu.Lock()
                        next := b.next
                        b.count++
                        b.mu.Unlock()
                        if !next.IsZero() {
                                delay := next.Sub(b.nowFn())
                                if delay > 0 {
                                        if b.jitter {
                                                delay += time.Duration(rand.Int64N(int64(delay))) /*#nosec G404*/
                                        }
                                        err = b.sleepFn(ctx, delay)
                                }
                        }
                }
        }
        return
}

func (b *logStreamBackoff) success() {
        if b != nil {
                b.mu.Lock()
                if b.factor > 0 {
                        b.current = time.Duration(float64(b.current) / b.factor)
                        if b.current < b.min {
                                b.current = 0
                        }
                }
                if b.current > 0 {
                        b.next = b.nowFn().Add(b.current)
                } else {
                        b.next = time.Time{}
                }
                b.mu.Unlock()
        }
}

func (b *logStreamBackoff) backoff() {
        if b != nil {
                b.mu.Lock()
                defer b.mu.Unlock()

                if b.current > 0 && b.factor > 0 {
                        b.current = time.Duration(float64(b.current) * b.factor)
                }
                b.current = max(b.min, b.current)
                b.current = min(b.max, b.current)
                b.next = time.Time{}
                if b.current > 0 {
                        b.next = b.nowFn().Add(b.current)
                }
        }
}

func rawEntriesStopIndex(start, end int64) (stop int64) {
        stop = start
        if start <= end {
                // End each window at a common boundary so truncated responses can
                // bring the next request onto a cacheable full-size range.
                stop = start + min(end-start+1, LogBatchSize-start%LogBatchSize) - 1
        }
        return
}

// getRawEntries fetches and processes the logentries in the index range start...end (inclusive) in order with no parallelism.
// Returns 'wanted' set to true if handleFn returned true for any logentry.
// Returns an error if not all entries could be fetched and processed.
func (ls *LogStream) getRawEntries(ctx context.Context, client rawEntriesClient, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (wanted bool, next int64, err error) {
        next = start
        for next <= end && err == nil {
                stopIndex := rawEntriesStopIndex(next, end)
                var resp *ct.GetEntriesResponse
                if err = ls.backoff.Retry(ctx, func() (e error) {
                        if ls.LogOperator != nil {
                                if sleeptime := min(500, ls.Status429.Load()*2); sleeptime > 0 {
                                        if historical {
                                                sleeptime *= 2
                                        }
                                        _ = sleep(ctx, time.Millisecond*time.Duration(sleeptime))
                                }
                        }
                        if e = ls.waitForQueue(ctx, historical); e != nil {
                                return
                        }
                        if resp, e = client.GetRawEntries(ctx, next, stopIndex); e != nil {
                                if !ls.handleStreamError(e, "GetRawEntries") {
                                        e = wrapLogStreamRetryable(e)
                                }
                        }
                        return
                }); err == nil {
                        now := time.Now()
                        for i := range resp.Entries {
                                if historical {
                                        if err = ls.waitForQueue(ctx, true); err != nil {
                                                break
                                        }
                                        now = time.Now()
                                }
                                le := ls.makeLogEntry(next, resp.Entries[i], historical)
                                if handleFn(ctx, now, le) {
                                        wanted = true
                                }
                                ls.seeIndex(next)
                                next++
                                if gapcounter != nil {
                                        gapcounter.Add(-1)
                                }
                        }
                }
                if err != nil {
                        if ctx.Err() == nil {
                                _ = ls.LogError(err, "GetRawEntries", "url", ls.URL(), "start", next, "end", end)
                        }
                        if gapcounter != nil {
                                gapcounter.Add(next - (end + 1))
                        }
                }
        }
        return
}

// getRawEntriesParallel fetches and processes the logentries in the index range start...end (inclusive) using Config.Concurrency workers.
// The returned next start index can be less than end+1 if an error occured.
// Returns the next start index and 'wanted' set to true if handleFn returned true for any logentry.
func (ls *LogStream) getRawEntriesParallel(ctx context.Context, client rawEntriesClient, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (wanted bool, next int64, err error) {
        type rawEntriesRange struct {
                start int64
                end   int64
        }

        err = ctx.Err()
        next = start

        sleepCtx, sleepCancel := context.WithCancel(ctx)
        workCh := make(chan rawEntriesRange)
        workerCount := min(32, max(1, ls.Concurrency))
        workerSleep := time.Second / time.Duration(workerCount)
        completed := make(map[int64]int64)
        var wg sync.WaitGroup
        var workMu sync.Mutex
        var stop atomic.Bool
        for i := range workerCount {
                wg.Add(1)
                go func(i int) {
                        defer wg.Done()
                        if i > 0 && ls.LogOperator != nil {
                                if ls.Status429.Load() > 0 {
                                        _ = sleep(sleepCtx, workerSleep*time.Duration(i))
                                }
                        }
                        for r := range workCh {
                                w, _, e := ls.getRawEntries(ctx, client, r.start, r.end, historical, handleFn, gapcounter)
                                workMu.Lock()
                                wanted = wanted || w
                                if e == nil {
                                        completed[r.start] = r.end
                                advanceNext:
                                        if end, ok := completed[next]; ok {
                                                delete(completed, next)
                                                next = end + 1
                                                goto advanceNext
                                        }
                                } else {
                                        err = e
                                        stop.Store(true)
                                }
                                workMu.Unlock()
                        }
                }(i)
        }

        for start <= end && ctx.Err() == nil && !stop.Load() {
                stopIndex := rawEntriesStopIndex(start, end)
                select {
                case workCh <- rawEntriesRange{start: start, end: stopIndex}:
                        start = stopIndex + 1
                case <-ctx.Done():
                }
        }

        close(workCh)
        sleepCancel()
        wg.Wait()
        if err == nil {
                err = ctx.Err()
        }

        return
}

func (ls *LogStream) makeTileLogEntry(logindex int64, entry *sunlight.LogEntry, historical bool) (le *LogEntry) {
        le = &LogEntry{
                LogStream:  ls,
                LogIndex:   logindex,
                Historical: historical,
        }
        if entry != nil {
                le.Seen = time.UnixMilli(entry.Timestamp).UTC()
                if entry.IsPrecert {
                        le.PreCert = true
                }
                var cert *x509.Certificate
                var certErr error
                if entry.IsPrecert {
                        cert, certErr = x509.ParseTBSCertificate(entry.Certificate)
                } else {
                        cert, certErr = x509.ParseCertificate(entry.Certificate)
                }
                if certErr != nil {
                        le.Err = certErr
                }
                le.Certificate = cert
                if entry.IsPrecert && len(entry.PreCertificate) > 0 {
                        shasig := sha256.Sum256(entry.PreCertificate)
                        le.Signature = shasig[:]
                } else if len(entry.Certificate) > 0 {
                        shasig := sha256.Sum256(entry.Certificate)
                        le.Signature = shasig[:]
                }
        }
        if len(le.Signature) == 0 && le.Certificate != nil {
                if raw := le.Certificate.Raw; len(raw) > 0 {
                        shasig := sha256.Sum256(raw)
                        le.Signature = shasig[:]
                } else if raw := le.Certificate.RawTBSCertificate; len(raw) > 0 {
                        shasig := sha256.Sum256(raw)
                        le.Signature = shasig[:]
                }
        }
        return
}

func (ls *LogStream) getTileEntries(ctx context.Context, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (next int64, wanted bool) {
        client := ls.headTile
        if historical && ls.tailTile != nil {
                client = ls.tailTile
        }
        return ls.getTileEntriesWithClient(ctx, client, start, end, historical, handleFn, gapcounter)
}

func (ls *LogStream) getTileEntriesWithClient(ctx context.Context, client *sunlight.Client, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (next int64, wanted bool) {
        next = start
        for next <= end {
                entryErr := ls.backoff.Retry(ctx, func() (err error) {
                        if err = ls.waitForQueue(ctx, historical); err != nil {
                                return
                        }
                        var checkpoint torchwood.Checkpoint
                        if checkpoint, _, err = client.Checkpoint(ctx); err == nil {
                                if next >= checkpoint.N {
                                        return wrapLogStreamRetryable(ErrSTHDiffTooLow)
                                }
                                processEntry := func(logindex int64, entry *sunlight.LogEntry) {
                                        le := ls.makeTileLogEntry(logindex, entry, historical)
                                        ls.seeIndex(logindex)
                                        if handleFn(ctx, time.Now(), le) {
                                                wanted = true
                                        }
                                        next = logindex + 1
                                        if gapcounter != nil {
                                                gapcounter.Add(-1)
                                        }
                                }
                                if err = ls.waitForQueue(ctx, historical); err != nil {
                                        return
                                }
                                if next == end {
                                        var entry *sunlight.LogEntry
                                        if entry, _, err = client.Entry(ctx, checkpoint.Tree, next); err == nil {
                                                processEntry(next, entry)
                                        }
                                } else {
                                        for logindex, entry := range client.Entries(ctx, checkpoint.Tree, next) {
                                                if err = ls.waitForQueue(ctx, historical); err != nil {
                                                        break
                                                }
                                                processEntry(logindex, entry)
                                                if logindex >= end {
                                                        break
                                                }
                                        }
                                        if err == nil {
                                                err = client.Err()
                                        }
                                }
                        }
                        if err != nil {
                                if !ls.handleStreamError(err, "getTileEntries") {
                                        err = wrapLogStreamRetryable(err)
                                }
                        }
                        return
                })
                if entryErr != nil {
                        if gapcounter != nil {
                                if ctx.Err() == nil {
                                        _ = ls.LogError(entryErr, "gap not fillable", "url", ls.URL(), "start", next, "end", end)
                                }
                                gapcounter.Add(next - (end + 1))
                        }
                        break
                }
        }
        return
}

// OperatorDomain returns the TLD+1 given an URL.
func OperatorDomain(urlString string) string {
        opDom := urlString
        if u, err := url.Parse(urlString); err == nil {
                opDom = u.Hostname()
                if idx := strings.LastIndexByte(opDom, '.'); idx > 0 {
                        if idx := strings.LastIndexByte(opDom[:idx], '.'); idx > 0 {
                                opDom = opDom[idx+1:]
                        }
                }
        }
        return opDom
}

var (
        // ErrLogEntriesTooOld reports that historical entries exceed the configured maximum age.
        ErrLogEntriesTooOld = errors.New("log entries are older than max age")
        errGapIncomplete    = errors.New("gap fetch incomplete")
)

type errLogEntriesTooOld struct {
        MaxAge int
}

type rawEntriesFetcher func(ctx context.Context, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (next int64, wanted bool)

func (ls *LogStream) getGapEntries(ctx context.Context, start, end int64, historical bool, handleFn handleLogEntryFn, gapcounter *atomic.Int64) (next int64, wanted bool) {
        if ls.gapTile != nil {
                return ls.getTileEntriesWithClient(ctx, ls.gapTile, start, end, historical, handleFn, gapcounter)
        }
        return ls.getEntries(ctx, start, end, historical, handleFn, gapcounter)
}

func (e errLogEntriesTooOld) Error() string {
        return "log entries are older than " + strconv.Itoa(e.MaxAge) + " days"
}

func (e errLogEntriesTooOld) Unwrap() error {
        return ErrLogEntriesTooOld
}

func (cdb *PgDB) backfillGapsWithFetcher(ctx context.Context, ls *LogStream, fetchFn rawEntriesFetcher, scanDone <-chan error) (scanErr error) {
        return cdb.backfillGapsWithFetcherMode(ctx, ls, fetchFn, scanDone, true)
}

func (cdb *PgDB) backfillGapsWithFetcherMode(ctx context.Context, ls *LogStream, fetchFn rawEntriesFetcher, scanDone <-chan error, includeTrailing bool) (scanErr error) {
        if cdb != nil && ls != nil {
                var lastgap gap
                var maxErr error
                if includeTrailing {
                        var nullableMaxIndex sql.NullInt64
                        lastindex := ls.LastIndex.Load()
                        row := cdb.QueryRow(ctx, cdb.stmtSelectMaxIdx, ls.Id)
                        maxErr = cdb.LogError(row.Scan(&nullableMaxIndex), "backfillGapsWithFetcher/MaxIndex", "url", ls.URL())
                        if maxErr == nil && nullableMaxIndex.Valid {
                                ls.seeIndex(nullableMaxIndex.Int64)
                                if lastindex < nullableMaxIndex.Int64 {
                                        lastindex = nullableMaxIndex.Int64
                                }
                                if nullableMaxIndex.Int64 < lastindex {
                                        lastgap = gap{start: nullableMaxIndex.Int64 + 1, end: lastindex}
                                        cdb.recordGap(ls, lastgap)
                                }
                        }
                }
                scanErr = maxErr
                if gapCh := ls.getGapCh(); gapCh != nil {
                        allGapsFilled := maxErr == nil
                        fillGap := func(g gap) {
                                if ctx.Err() == nil {
                                        ls.Backfill.Add((g.end - g.start) + 1)
                                        next, _ := fetchFn(ctx, g.start, g.end, true, ls.sendEntry, &ls.Backfill)
                                        // Fetched entries enter a queue; only a later DB scan proves they were stored.
                                        if next < g.end+1 {
                                                allGapsFilled = false
                                        }
                                } else {
                                        allGapsFilled = false
                                }
                        }
                        for gap := range gapCh {
                                fillGap(gap)
                        }
                        if scanDone != nil {
                                scanErr = errors.Join(scanErr, <-scanDone)
                                if scanErr != nil {
                                        allGapsFilled = false
                                }
                        }
                        if lastgap.end != 0 {
                                fillGap(lastgap)
                        }
                        if !allGapsFilled && scanErr == nil && ctx.Err() == nil {
                                scanErr = errGapIncomplete
                        }
                }
        }
        return
}

func (cdb *PgDB) scanAndBackfillGaps(ctx context.Context, ls *LogStream, fetchFn rawEntriesFetcher) (scanErr error) {
        return cdb.scanAndBackfillGapsMode(ctx, ls, fetchFn, true)
}

func (cdb *PgDB) scanAndBackfillGapsMode(ctx context.Context, ls *LogStream, fetchFn rawEntriesFetcher, includeTrailing bool) (scanErr error) {
        started := time.Now()
        ctx, cancel := context.WithTimeout(ctx, time.Hour)
        defer cancel()

        gapCh := make(chan gap, 8)
        ls.mu.Lock()
        ls.gapCh = gapCh
        ls.mu.Unlock()

        scanDone := make(chan error, 1)
        go func() {
                scanDone <- cdb.selectStreamGaps(ctx, ls)
        }()
        scanErr = cdb.backfillGapsWithFetcherMode(ctx, ls, fetchFn, scanDone, includeTrailing)

        ls.mu.Lock()
        if ls.gapCh == gapCh {
                ls.gapCh = nil
        }
        ls.mu.Unlock()
        if err := ctx.Err(); err != nil {
                scanErr = errors.Join(scanErr, err)
        }
        if scanErr == nil {
                cdb.LogInfo("gap scan completed", "url", ls.URL(), "stream", ls.Id, "elapsed", time.Since(started).Round(time.Millisecond))
        }
        return
}

func (cdb *PgDB) rescanGaps(ctx context.Context, ls *LogStream, next time.Time, wg *sync.WaitGroup) {
        defer wg.Done()
        for ctx.Err() == nil {
                if sleep(ctx, time.Until(next)) != nil {
                        return
                }
                _ = cdb.scanAndBackfillGapsMode(ctx, ls, ls.getGapEntries, false)
                if err := cdb.recoverTrailingGap(ctx, ls, ls.getGapEntries); err != nil && !errors.Is(err, errGapScanDeferred) {
                        _ = cdb.LogError(err, "recoverTrailingGap", "url", ls.URL())
                }
                next = time.Now().Add(gapScanInterval)
        }
}

func (cdb *PgDB) recoverTrailingGap(ctx context.Context, ls *LogStream, fetchFn rawEntriesFetcher) (err error) {
        if cdb.QueueUsage() > 50 {
                return errGapScanDeferred
        }
        queryCtx, cancel := context.WithTimeout(ctx, gapQueryTimeout)
        defer cancel()
        var row sql.NullInt64
        if err = cdb.QueryRow(queryCtx, cdb.stmtSelectMaxIdx, ls.Id).Scan(&row); err != nil {
                return
        }
        storedMax := row.Int64
        if !row.Valid {
                if firstFetched := ls.MinIndex.Load(); firstFetched >= 0 {
                        storedMax = firstFetched - 1
                } else {
                        return
                }
        }
        head := ls.MaxIndex.Load()
        ls.mu.Lock()
        stable := ls.tailCheckValid && ls.tailCheckMax == storedMax
        ls.tailCheckMax = storedMax
        ls.tailCheckValid = true
        ls.mu.Unlock()
        if !stable || head <= storedMax {
                return
        }
        end := min(head, storedMax+gapWindowSize)
        ls.Backfill.Add(end - storedMax)
        next, _ := fetchFn(ctx, storedMax+1, end, true, ls.sendEntry, &ls.Backfill)
        if next <= end {
                err = errGapIncomplete
        }
        return
}

// Return the minimum stored log index, or -1 when the stream has no entries.
func (cdb *PgDB) backfillMinIndex(ctx context.Context, ls *LogStream) (minIndex int64, err error) {
        minIndex = -1
        var minIndexRow sql.NullInt64
        row := cdb.QueryRow(ctx, cdb.stmtSelectMinIdx, ls.Id)
        if err = cdb.LogError(row.Scan(&minIndexRow), "Backfill/MinIndex", "url", ls.URL()); err == nil {
                if minIndexRow.Valid {
                        minIndex = minIndexRow.Int64
                        ls.seeIndex(minIndex)
                }
        }
        return
}

func (cdb *PgDB) backfillSetGapStartIndex(ctx context.Context, ls *LogStream, logindex int64) (err error) {
        if cdb != nil && ls != nil {
                if ctx.Err() == nil {
                        if logindex >= 0 {
                                _, err = cdb.Exec(ctx, cdb.stmtUpdateBackfillIdx, logindex, ls.Id)
                                err = cdb.LogError(err, "Backfill/UpdateIndex", "url", ls.URL(), "stream", ls.Id, "logindex", logindex)
                        }
                }
        }
        return
}

func reverseBackfillStart(minIndex int64, tiled bool) int64 {
        if tiled {
                // Torchwood reads up to 50 whole data tiles at once. Keep reverse
                // windows on those boundaries so each tile is fetched only once.
                const window = 50 * sunlight.TileWidth
                return (minIndex - 1) / window * window
        }
        // The age cutoff is decided once per reverse window. Keep at least
        // 1024 entries in each window when available, then align its start.
        window := max(int64(1024), LogBatchSize)
        return max(0, (minIndex-window)/LogBatchSize*LogBatchSize)
}

func (cdb *PgDB) backfillStream(ctx context.Context, ls *LogStream, wg *sync.WaitGroup) {
        defer wg.Done()
        var minIndex int64
        var err error

        _ = cdb.scanAndBackfillGaps(ctx, ls, ls.getGapEntries)
        var scanWG sync.WaitGroup
        scanWG.Add(1)
        go cdb.rescanGaps(ctx, ls, time.Now().Add(gapScanInterval), &scanWG)
        defer scanWG.Wait()
        for ctx.Err() == nil {
                if minIndex, err = cdb.backfillMinIndex(ctx, ls); err == nil {
                        break
                }
                if sleep(ctx, time.Minute) != nil {
                        return
                }
        }
        if minIndex < 0 {
                // An empty database still needs the entries preceding the known head.
                minIndex = ls.LastIndex.Load() + 1
        }
        if err == nil && minIndex > 0 {
                cdb.LogInfo("backlog start", "url", ls.URL(), "stream", ls.Id, "logindex", minIndex)
                ls.Backfill.Add(minIndex)
                for minIndex > 0 && ctx.Err() == nil {
                        start := reverseBackfillStart(minIndex, ls.isTiled())
                        stop := minIndex - 1
                        next := start
                        wanted := false
                        retryDelay := time.Second
                        for next <= stop && ctx.Err() == nil {
                                fetched, windowWanted := ls.getEntries(ctx, next, stop, true, ls.sendEntry, &ls.Backfill)
                                wanted = wanted || windowWanted
                                if fetched <= stop {
                                        if ctx.Err() != nil {
                                                return
                                        }
                                        // Fetchers remove the unfetched remainder from Backfill on error.
                                        ls.Backfill.Add(stop + 1 - fetched)
                                        if fetched > next {
                                                retryDelay = time.Second
                                        } else {
                                                retryDelay = min(retryDelay*2, 30*time.Second)
                                        }
                                        if sleep(ctx, retryDelay) != nil {
                                                return
                                        }
                                }
                                next = fetched
                        }
                        if ctx.Err() != nil {
                                return
                        }
                        minIndex = start
                        if !wanted {
                                ls.addError(ls, errLogEntriesTooOld{MaxAge: cdb.PgMaxAge})
                                ls.Backfill.Add(-minIndex)
                                break
                        }
                }
                if ctx.Err() == nil {
                        cdb.LogInfo("backlog stops", "url", ls.URL(), "stream", ls.Id, "logindex", minIndex)
                }
        }
}

type gapCounts struct {
        url           string
        gaps, entries int64
}

func (cdb *PgDB) recordGap(ls *LogStream, g gap) {
        length := (g.end - g.start) + 1
        url := ls.URL()
        cdb.mu.Lock()
        if cdb.gaps == nil {
                cdb.gaps = make(map[int32]gapCounts)
        }
        counts := cdb.gaps[ls.Id]
        counts.url = url
        counts.gaps++
        counts.entries += length
        cdb.gaps[ls.Id] = counts
        cdb.mu.Unlock()
        if length > 1000 {
                cdb.LogInfo("gap", "url", url, "stream", ls.Id, "logindex", g.start, "length", length)
        }
}

func (cdb *PgDB) logGaps() {
        cdb.mu.Lock()
        pending := cdb.gaps
        cdb.gaps = nil
        cdb.mu.Unlock()
        // Counts survive reader replacement or removal until they are reported.
        for id, counts := range pending {
                cdb.LogInfo("gaps", "url", counts.url, "stream", id, "gaps", counts.gaps, "entries", counts.entries)
        }
}

func (cdb *PgDB) reportGaps(ctx context.Context, wg *sync.WaitGroup) {
        defer wg.Done()
        ticker := time.NewTicker(time.Minute)
        defer ticker.Stop()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                        cdb.logGaps()
                }
        }
}

func (cdb *PgDB) runBatch(ctx context.Context, queued []*LogEntry) (err error) {
        var b []byte
        b = append(b, `[`...)
        for i, le := range queued {
                if i > 0 {
                        b = append(b, `,`...)
                }
                b = le.appendJSON(b)
        }
        b = append(b, `]`...)
        now := time.Now()
        _, err = cdb.Exec(ctx, cdb.funcIngestBatch, string(b))
        elapsed := time.Since(now)
        if err != nil {
                if pe, ok := err.(*pgconn.PgError); ok {
                        if pe.SQLState() == "22P02" {
                                cdb.LogInfo("generated invalid JSON data", "json", string(b))
                        }
                }
        }
        cdb.mu.Lock()
        cdb.newentrycount += int64(len(queued))
        cdb.newentrytime += elapsed
        cdb.mu.Unlock()
        return
}

func (cdb *PgDB) worker(ctx context.Context, wg *sync.WaitGroup, workerID int) {
        defer wg.Done()
        if batchCh := cdb.getBatchCh(workerID); batchCh != nil {
                cdb.Workers.Add(1)
                defer cdb.Workers.Add(-1)
                const tickerInterval = time.Second * 10
                staggerInterval := tickerInterval / time.Duration(cdb.workerCount)
                tckr := time.NewTicker(1 + (staggerInterval * time.Duration(workerID)))
                stop := false
                var queued []*LogEntry
                for !stop {
                        ticked := false
                        var flushAck chan struct{}
                        select {
                        case <-ctx.Done():
                                stop = true
                        case <-tckr.C:
                                ticked = true
                                if staggerInterval != 0 {
                                        staggerInterval = 0
                                        tckr.Stop()
                                        tckr = time.NewTicker(tickerInterval)
                                        defer tckr.Stop()
                                }
                        case le, ok := <-batchCh:
                                if !ok {
                                        stop = true
                                } else if le.flushAck != nil {
                                        flushAck = le.flushAck
                                        ticked = true
                                } else {
                                        queued = append(queued, le)
                                }
                        }
                        if l := len(queued); l > 0 && (l >= DbIngestBatchSize || ticked || stop) {
                                _ = cdb.LogError(cdb.runBatch(ctx, queued), "runBatch")
                                if flushAck != nil {
                                        flushAck <- struct{}{}
                                        flushAck = nil
                                }
                                for _, le := range queued {
                                        if !stop {
                                                select {
                                                case <-ctx.Done():
                                                        stop = true
                                                case cdb.getSendEntryCh() <- le:
                                                }
                                        }
                                }
                                clear(queued)
                                queued = queued[:0]
                        }
                        if flushAck != nil {
                                flushAck <- struct{}{}
                        }
                }
        }
}

func (cdb *PgDB) AverageNewEntryTime() (d time.Duration) {
        cdb.mu.Lock()
        d = cdb.avgentrytime
        cdb.mu.Unlock()
        return
}

func (cdb *PgDB) runWorkers(ctx context.Context, wg *sync.WaitGroup) {
        const interval = time.Millisecond * 100
        defer wg.Done()

        wg.Add(cdb.workerCount)
        for i := 0; i < cdb.workerCount; i++ {
                go cdb.worker(ctx, wg, i)
        }

        ticks := 0
        ticker := time.NewTicker(interval)
        avgentrytimes := make([]time.Duration, time.Minute/interval)
        defer ticker.Stop()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                        cdb.mu.Lock()
                        avgentrytime := cdb.newentrytime
                        if d := time.Duration(cdb.newentrycount); d > 0 {
                                avgentrytime /= d
                        }
                        cdb.newentrytime = 0
                        cdb.newentrycount = 0
                        avgentrytimes[ticks] = avgentrytime
                        ticks++
                        if ticks >= cap(avgentrytimes) {
                                ticks = 0
                        }
                        avgentrytime = 0
                        for _, d := range avgentrytimes {
                                avgentrytime += d
                        }
                        cdb.avgentrytime = avgentrytime / time.Duration(cap(avgentrytimes))
                        cdb.mu.Unlock()
                }
        }
}

// NewPgDB creates a PgDB and creates the needed tables and indices if they don't exist.
func NewPgDB(ctx context.Context, cs *CertStream) (cdb *PgDB, err error) {
        const callOperatorID = `SELECT CERTDB_operator_id($1,$2);`
        const callStreamID = `SELECT CERTDB_stream_id($1,$2,$3);`

        if cs.PgAddr != "" {
                dsn := fmt.Sprintf("postgres://%s:%s@%s/%s?pool_max_conns=%d&pool_max_conn_idle_time=1m",
                        cs.PgUser, cs.PgPass, cs.PgAddr, cs.PgName, cs.PgConns)
                if !cs.PgSyncCommit {
                        dsn += "&synchronous_commit=off"
                }
                if cs.PgNoSSL {
                        dsn += "&sslmode=disable"
                }
                var poolcfg *pgxpool.Config
                if poolcfg, err = pgxpool.ParseConfig(dsn); err == nil {
                        poolcfg.ConnConfig.DefaultQueryExecMode = pgx.QueryExecModeSimpleProtocol
                        var pool *pgxpool.Pool
                        if pool, err = pgxpool.NewWithConfig(ctx, poolcfg); err == nil {
                                if err = pool.Ping(ctx); err == nil {
                                        cs.LogInfo("database", "addr", cs.PgAddr, "name", cs.PgName, "prefix", cs.PgPrefix)
                                        pfx := func(s string) string { return strings.ReplaceAll(s, "CERTDB_", cs.PgPrefix) }
                                        if err = ensureSchema(ctx, pool, pfx); err == nil {
                                                var pgversion string
                                                if cs.LogError(pool.QueryRow(ctx, `SELECT version();`).Scan(&pgversion), "postgres version") == nil {
                                                        cs.LogInfo("postgres", "version", pgversion)
                                                }
                                                workerBits := min(8, max(1, cs.PgWorkerBits))
                                                workerCount := 1 << workerBits
                                                batchChans := make([]chan *LogEntry, workerCount)
                                                for i := range batchChans {
                                                        batchChans[i] = make(chan *LogEntry, DbIngestBatchSize*2)
                                                }
                                                cdb = &PgDB{
                                                        CertStream:            cs,
                                                        Pool:                  pool,
                                                        Pfx:                   pfx,
                                                        funcOperatorID:        pfx(callOperatorID),
                                                        funcStreamID:          pfx(callStreamID),
                                                        funcIngestBatch:       pfx(`SELECT CERTDB_ingest_batch($1::jsonb);`),
                                                        stmtSelectGapWindow:   pfx(selectGapWindow),
                                                        stmtSelectMinIdx:      pfx(SelectMinIndex),
                                                        stmtSelectMaxIdx:      pfx(SelectMaxIndex),
                                                        stmtSelectBackfillIdx: pfx(SelectBackfillIndex),
                                                        stmtUpdateBackfillIdx: pfx(UpdateBackfillIndex),
                                                        stmtLowerBackfillIdx:  pfx(lowerBackfillIndex),
                                                        batchCh:               batchChans,
                                                        gapScanPermit:         make(chan struct{}, 1),
                                                        workerBits:            workerBits,
                                                        workerCount:           workerCount,
                                                        estimates: map[string]float64{
                                                                "cert":   0,
                                                                "domain": 0,
                                                                "entry":  0,
                                                        },
                                                }
                                                cdb.refreshEstimates(ctx)
                                        }
                                }
                        }
                }
        }
        if cdb != nil {
                cs.LogInfo("database workers", "count", cdb.workerCount, "bits", cdb.workerBits)
        }
        return
}

func (cdb *PgDB) beginCall() {
        if cdb != nil {
                cdb.closeMu.RLock()
                cdb.closeWG.Add(1)
                cdb.closeMu.RUnlock()
        }
}

func (cdb *PgDB) endCall() {
        if cdb != nil {
                cdb.closeWG.Done()
        }
}

func (cdb *PgDB) Close() {
        if cdb != nil {
                cdb.closeOnce.Do(func() {
                        cdb.closeMu.Lock()
                        cdb.mu.Lock()
                        chans := cdb.batchCh
                        cdb.batchCh = nil
                        cdb.mu.Unlock()
                        for _, ch := range chans {
                                if ch != nil {
                                        close(ch)
                                }
                        }
                        cdb.closeWG.Wait()
                        cdb.Pool.Close()
                        cdb.closeMu.Unlock()
                })
        }
}

func (cdb *PgDB) QueueUsage() (pct int) {
        cdb.mu.Lock()
        chans := cdb.batchCh
        cdb.mu.Unlock()
        totalLen := 0
        totalCap := 0
        for _, ch := range chans {
                if ch != nil {
                        totalLen += len(ch)
                        totalCap += cap(ch)
                }
        }
        if totalCap > 0 {
                pct = totalLen * 100 / totalCap
        }
        return
}

func (cdb *PgDB) getBatchCh(idx int) (ch chan *LogEntry) {
        cdb.mu.Lock()
        ch = cdb.batchCh[idx]
        cdb.mu.Unlock()
        return
}

func (cdb *PgDB) workerIndexFor(le *LogEntry) (idx int) {
        if cdb != nil && le != nil {
                if cert := le.Cert(); cert != nil && len(cert.Signature) > 0 {
                        idx = int(cert.Signature[0] >> (8 - cdb.workerBits))
                }
        }
        return
}

func (cdb *PgDB) sendToBatcher(ctx context.Context, le *LogEntry) {
        if le != nil && ctx.Err() == nil {
                if ch := cdb.getBatchCh(cdb.workerIndexFor(le)); ch != nil {
                        select {
                        case <-ctx.Done():
                        case ch <- le:
                        }
                }
        }
}

func (cdb *PgDB) flushWorkers(ctx context.Context) (err error) {
        if cdb == nil {
                return
        }
        ctx, cancel := context.WithTimeout(ctx, 30*time.Second)
        defer cancel()
        cdb.closeMu.RLock()
        defer cdb.closeMu.RUnlock()
        cdb.mu.Lock()
        chans := cdb.batchCh
        cdb.mu.Unlock()
        acks := make(chan struct{}, len(chans))
        for _, ch := range chans {
                select {
                case <-ctx.Done():
                        return ctx.Err()
                case ch <- &LogEntry{flushAck: acks}:
                }
        }
        for range chans {
                select {
                case <-ctx.Done():
                        return ctx.Err()
                case <-acks:
                }
        }
        return
}

func RenderSQL(query string, args ...any) string {
        for i, arg := range args {
                var s string
                switch v := arg.(type) {
                case string:
                        s = fmt.Sprintf("'%s'", strings.ReplaceAll(v, "'", "''"))
                case time.Time:
                        s = fmt.Sprintf("'%s'", v.Format(time.RFC3339))
                default:
                        s = fmt.Sprint(v)
                }
                query = strings.ReplaceAll(query, fmt.Sprintf("$%d", i+1), s)
        }
        return query
}

// CleanEmptyCheckTimeout bounds each scan [PgDB.DeleteStream] makes over a
// stream's already deleted entries: restarting the cursor at the first entry,
// and confirming the stream holds none before deleting it.
//
// A drained stream keeps its index entries until autovacuum removes them, and
// on a large table walking them to find that none are live takes minutes. When
// the check runs out of time the stream is left in place for a later call
// rather than deleted without proof.
var CleanEmptyCheckTimeout = 30 * time.Second

// DeleteStream deletes up to batchSize log entries belonging to the stream,
// oldest first, and deletes the stream itself once it holds none.
//
// Deletion resumes from CERTDB_stream.clean_logindex, so repeated calls walk
// forward in logindex order instead of rescanning the index entries of rows
// already deleted, and a restart picks up where the last one stopped rather
// than from the first entry. Reaching the end of the stream restarts the scan
// at its first entry, so entries backfilled below the cursor are still found.
//
// The stream itself is deleted only once it holds no entries at all, checked
// under a lock on the stream row. A batch deleting nothing is not on its own
// proof of that, since rows locked by another deleter are skipped.
//
// A batchSize below one deletes nothing.
func (cdb *PgDB) DeleteStream(ctx context.Context, streamId int32, batchSize int) (rowsDeleted int64, err error) {
        if cdb != nil {
                cdb.beginCall()
                defer cdb.endCall()
                if batchSize > 0 {
                        query := cdb.Pfx(`WITH CERTDB_clean_stream AS (
  SELECT ctid
  FROM CERTDB_entry
  WHERE stream = $1 AND logindex >= $2
  ORDER BY logindex ASC
  LIMIT $3
  FOR UPDATE SKIP LOCKED
), CERTDB_deleted_entry AS (
  DELETE FROM CERTDB_entry
  USING CERTDB_clean_stream
  WHERE CERTDB_entry.ctid = CERTDB_clean_stream.ctid
  RETURNING CERTDB_entry.logindex
)
SELECT count(*), max(logindex) FROM CERTDB_deleted_entry;`)
                        deleteBatch := func(batchCtx context.Context, from int64) (deleted int64, next int64, batchErr error) {
                                var highest sql.NullInt64
                                if batchErr = cdb.QueryRow(batchCtx, query, streamId, from, batchSize).Scan(&deleted, &highest); batchErr == nil {
                                        if highest.Valid {
                                                next = highest.Int64 + 1
                                        }
                                }
                                return
                        }
                        var from int64
                        if err = cdb.QueryRow(ctx, cdb.Pfx(`SELECT clean_logindex FROM CERTDB_stream WHERE id = $1;`), streamId).Scan(&from); err == nil {
                                var next int64
                                if rowsDeleted, next, err = deleteBatch(ctx, from); err == nil {
                                        unproven := false
                                        if rowsDeleted == 0 && from != 0 {
                                                // Nothing above the cursor, so look again from the start
                                                // in case backfill added entries below it. That walks
                                                // the entries already deleted from this stream, the
                                                // same unbounded scan deleteEmptyStream faces, so give
                                                // it the same budget.
                                                restartCtx, cancel := context.WithTimeout(ctx, CleanEmptyCheckTimeout)
                                                rowsDeleted, next, err = deleteBatch(restartCtx, 0)
                                                cancel()
                                                if errors.Is(err, context.DeadlineExceeded) {
                                                        rowsDeleted, next, err, unproven = 0, from, ctx.Err(), true
                                                }
                                        }
                                        if err == nil && next != from {
                                                _, err = cdb.Exec(ctx, cdb.Pfx(`UPDATE CERTDB_stream SET clean_logindex = $2 WHERE id = $1;`), streamId, next)
                                        }
                                        // Only worth asking whether the stream is empty when the
                                        // scan that would have found entries actually finished.
                                        if err == nil && rowsDeleted == 0 && !unproven {
                                                rowsDeleted, err = cdb.deleteEmptyStream(ctx, streamId)
                                        }
                                }
                        } else if errors.Is(err, pgx.ErrNoRows) {
                                err = nil
                        }
                }
        }
        return
}

// deleteEmptyStream deletes the stream, and reports how many rows that removed,
// but only once it holds no entries.
//
// The stream row is locked before the check, so that inserting an entry, which
// takes a FOR KEY SHARE lock on that row for the foreign key, cannot commit
// between the check and the delete. Testing inside a single
// DELETE ... WHERE NOT EXISTS is not enough: the subquery sees the statement
// snapshot, so an entry committed after that snapshot is taken is invisible to
// it and still removed by the cascade.
func (cdb *PgDB) deleteEmptyStream(ctx context.Context, streamId int32) (rowsDeleted int64, err error) {
        var tx pgx.Tx
        if tx, err = cdb.Begin(ctx); err == nil {
                defer func() {
                        _ = tx.Rollback(ctx)
                }()
                var lockedId int32
                if err = tx.QueryRow(ctx, cdb.Pfx(`SELECT id FROM CERTDB_stream WHERE id = $1 FOR UPDATE;`), streamId).Scan(&lockedId); err == nil {
                        checkCtx, cancel := context.WithTimeout(ctx, CleanEmptyCheckTimeout)
                        defer cancel()
                        var hasEntries bool
                        err = tx.QueryRow(checkCtx, cdb.Pfx(`SELECT EXISTS (SELECT 1 FROM CERTDB_entry WHERE stream = $1);`), streamId).Scan(&hasEntries)
                        switch {
                        case errors.Is(err, context.DeadlineExceeded):
                                // Too many index entries still to walk to prove it empty.
                                // Leave the stream alone; a later call will be cheaper once
                                // autovacuum has removed them.
                                err = ctx.Err()
                        case err != nil:
                        case hasEntries:
                                err = tx.Commit(ctx)
                        default:
                                var tag pgconn.CommandTag
                                if tag, err = tx.Exec(ctx, cdb.Pfx(`DELETE FROM CERTDB_stream WHERE id = $1;`), streamId); err == nil {
                                        rowsDeleted = tag.RowsAffected()
                                        err = tx.Commit(ctx)
                                }
                        }
                } else if errors.Is(err, pgx.ErrNoRows) {
                        err = nil
                }
        }
        return
}

func (cdb *PgDB) Estimate(table string) (f float64) {
        table = strings.TrimPrefix(table, "CERTDB_")
        table = strings.TrimPrefix(table, cdb.PgPrefix)
        cdb.mu.Lock()
        f = cdb.estimates[table]
        cdb.mu.Unlock()
        return
}

func (cdb *PgDB) refreshEstimatesBatch() (batch *pgx.Batch) {
        batch = &pgx.Batch{}
        cdb.mu.Lock()
        defer cdb.mu.Unlock()
        for k := range cdb.estimates {
                table := cdb.Pfx("CERTDB_" + k)
                batch.Queue(SelectEstimate, table).QueryRow(func(row pgx.Row) error {
                        var estimate float64
                        if cdb.LogError(row.Scan(&estimate), "refreshEstimates", "table", table) == nil {
                                cdb.mu.Lock()
                                cdb.estimates[k] = estimate
                                cdb.mu.Unlock()
                        }
                        return nil
                })
        }
        return
}

func (cdb *PgDB) refreshEstimates(ctx context.Context) {
        if batch := cdb.refreshEstimatesBatch(); batch != nil {
                ctx, cancel := context.WithTimeout(ctx, time.Minute)
                defer cancel()
                _ = cdb.LogError(cdb.SendBatch(ctx, batch).Close(), "refreshEstimates")
        }
}

func (cdb *PgDB) estimator(ctx context.Context, wg *sync.WaitGroup) {
        ticker := time.NewTicker(time.Minute)
        defer func() {
                wg.Done()
                ticker.Stop()
        }()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                        cdb.refreshEstimates(ctx)
                }
        }
}

// CleanBatchSize limits the certificates automatic cleanup deletes per statement.
var CleanBatchSize = 10000

// CleanResetInterval is the pause after a completed certificate cleanup sweep.
//
// Each sweep revisits skipped locks and certificates inserted behind its position.
var CleanResetInterval = time.Hour

const cleanPagesPerBatch = 256

const deleteCertificatesPageQuery = `WITH CERTDB_clean_cert AS MATERIALIZED (
  SELECT ctid
  FROM CERTDB_cert
  WHERE ctid > $3::tid AND ctid < $4::tid AND notafter <= $1
  ORDER BY ctid
  LIMIT $2
  FOR UPDATE SKIP LOCKED
), CERTDB_deleted_cert AS (
  DELETE FROM CERTDB_cert
  WHERE ctid = ANY(ARRAY(SELECT ctid FROM CERTDB_clean_cert))
  RETURNING CERTDB_cert.ctid
)
SELECT count(*),
       (SELECT ctid FROM CERTDB_clean_cert ORDER BY ctid DESC LIMIT 1),
       pg_relation_size('CERTDB_cert'::regclass) / current_setting('block_size')::bigint
FROM CERTDB_deleted_cert;`

// DeleteCertificates deletes up to batchSize expired certificates in storage order.
//
// Certificates expiring at or before cutoff are eligible. Deletion cascades to
// their domain, IP address, email and URI rows; CT log entries are left alone.
// Each statement selects from a bounded range of adjacent heap pages. Empty ranges
// are crossed in separate transactions until a deletion or the end of the sweep.
// A sweep examines the whole heap, so sparse cleanup can still take a long time.
//
// Calls on the same PgDB serialize and resume the current sweep. Progress is
// checkpointed after the first processed range, once per minute thereafter, and
// on sweep completion. A new PgDB resumes the checkpoint; uncheckpointed work
// can be repeated. A checkpoint from a different physical heap is discarded.
// Concurrent PgDBs may repeat work through their shared checkpoint.
//
// Each sweep covers the heap size at its start. Later sweeps revisit certificates
// that expire, are inserted or move behind the cursor, as well as skipped locks
// and table rewrites. After a sweep completes, calls return zero until
// [CleanResetInterval] has passed. If saving a checkpoint fails, rowsDeleted can
// be positive with a non-nil error because deletion has already committed.
// A nil PgDB or a batchSize below one deletes nothing.
func (cdb *PgDB) DeleteCertificates(ctx context.Context, cutoff time.Time, batchSize int) (rowsDeleted int64, err error) {
        if cdb != nil {
                cdb.beginCall()
                defer cdb.endCall()
                if batchSize > 0 {
                        cdb.cleanMu.Lock()
                        defer cdb.cleanMu.Unlock()
                        if !cdb.cleanLoaded {
                                if err = cdb.loadCleanState(ctx); err != nil {
                                        return
                                }
                        }
                        if cdb.cleanEnd == 0 {
                                if !cdb.cleanResetAt.IsZero() && time.Since(cdb.cleanResetAt) < CleanResetInterval {
                                        if cdb.cleanSavedAt.Before(cdb.cleanResetAt) {
                                                err = cdb.saveCleanState(ctx, true)
                                        }
                                        return
                                }
                                if err = cdb.QueryRow(ctx, cdb.Pfx(`SELECT pg_relation_filenode('CERTDB_cert'::regclass),
pg_relation_size('CERTDB_cert'::regclass) / current_setting('block_size')::bigint;`)).Scan(&cdb.cleanFile, &cdb.cleanEnd); err != nil {
                                        return
                                }
                                cdb.cleanFrom = pgtype.TID{Valid: true}
                                cdb.cleanResetAt = time.Time{}
                                cdb.cleanSavedAt = time.Time{}
                        }
                        for cdb.cleanEnd > 0 && ctx.Err() == nil {
                                end := uint32(min(uint64(cdb.cleanEnd), uint64(cdb.cleanFrom.BlockNumber)+cleanPagesPerBatch))
                                upper := pgtype.TID{BlockNumber: end, Valid: true}
                                var last pgtype.TID
                                var blocks uint32
                                if err = cdb.QueryRow(ctx, cdb.Pfx(deleteCertificatesPageQuery), cutoff.UTC(), batchSize, cdb.cleanFrom, upper).Scan(&rowsDeleted, &last, &blocks); err != nil {
                                        return
                                }
                                // A full batch can stop within a page; keep its tuple offset.
                                if rowsDeleted == int64(batchSize) && last.Valid {
                                        cdb.cleanFrom = last
                                } else {
                                        cdb.cleanFrom = upper
                                }
                                cdb.cleanEnd = min(cdb.cleanEnd, blocks)
                                if cdb.cleanFrom.BlockNumber >= cdb.cleanEnd {
                                        cdb.cleanEnd = 0
                                        cdb.cleanResetAt = time.Now()
                                }
                                if err = cdb.saveCleanState(ctx, cdb.cleanEnd == 0); err != nil {
                                        return
                                }
                                if rowsDeleted > 0 {
                                        return
                                }
                        }
                        if err = ctx.Err(); err == nil && cdb.cleanEnd == 0 && cdb.cleanResetAt.IsZero() {
                                cdb.cleanResetAt = time.Now()
                                err = cdb.saveCleanState(ctx, true)
                        }
                }
        }
        return
}

func (cdb *PgDB) loadCleanState(ctx context.Context) (err error) {
        var blocks, savedFile uint32
        var completed pgtype.Timestamptz
        if err = cdb.QueryRow(ctx, cdb.Pfx(`SELECT pg_relation_filenode('CERTDB_cert'::regclass),
       pg_relation_size('CERTDB_cert'::regclass) / current_setting('block_size')::bigint,
       COALESCE(s.relfilenode, 0), COALESCE(s.cursor_tid, '(0,0)'::tid),
       COALESCE(s.end_block, 0), s.completed_at
FROM (SELECT 1) AS singleton
LEFT JOIN CERTDB_cert_cleanup s ON s.id = 1;`)).Scan(&cdb.cleanFile, &blocks, &savedFile, &cdb.cleanFrom, &cdb.cleanEnd, &completed); err == nil {
                if savedFile == cdb.cleanFile {
                        cdb.cleanEnd = min(cdb.cleanEnd, blocks)
                        if completed.Valid {
                                cdb.cleanResetAt = completed.Time
                                cdb.cleanSavedAt = completed.Time
                        }
                } else {
                        cdb.cleanFrom = pgtype.TID{Valid: true}
                        cdb.cleanEnd = blocks
                }
                cdb.cleanLoaded = true
        }
        return
}

func (cdb *PgDB) saveCleanState(ctx context.Context, force bool) (err error) {
        if force || cdb.cleanSavedAt.IsZero() || time.Since(cdb.cleanSavedAt) >= time.Minute {
                var completed *time.Time
                if !cdb.cleanResetAt.IsZero() {
                        completed = &cdb.cleanResetAt
                }
                // Keep the generation captured with the sweep. A rewrite makes this
                // checkpoint stale, so reopening the database starts at the new heap.
                if _, err = cdb.Exec(ctx, cdb.Pfx(`INSERT INTO CERTDB_cert_cleanup
    (id, relfilenode, cursor_tid, end_block, completed_at)
VALUES (1, $1, $2, $3, $4)
ON CONFLICT (id) DO UPDATE SET relfilenode = EXCLUDED.relfilenode,
    cursor_tid = EXCLUDED.cursor_tid, end_block = EXCLUDED.end_block,
    completed_at = EXCLUDED.completed_at;`), cdb.cleanFile, cdb.cleanFrom, cdb.cleanEnd, completed); err == nil {
                        cdb.cleanSavedAt = time.Now()
                }
        }
        return
}

// CleanCertificates deletes expired certificates until ctx is done.
//
// Certificates become eligible once Config.PgCertMaxAge has passed since
// expiry. A PgCertMaxAge of zero or less returns immediately without
// deleting anything. [Start] runs this for the lifetime of the CertStream,
// so callers need it only when driving a [PgDB] themselves.
func (cdb *PgDB) CleanCertificates(ctx context.Context) {
        maxAge := cdb.PgCertMaxAge
        if maxAge <= 0 {
                return
        }
        for ctx.Err() == nil {
                rowsDeleted, err := cdb.DeleteCertificates(ctx, time.Now().UTC().Add(-maxAge), CleanBatchSize)
                if cdb.LogError(err, "CleanCertificates") != nil || rowsDeleted == 0 {
                        select {
                        case <-ctx.Done():
                        case <-time.After(time.Minute):
                        }
                }
        }
}

const (
        gapWindowSize      int64 = 10_000
        gapRecentLookback  int64 = 1_000_000
        gapQueryTimeout          = 2 * time.Second
        gapRecentBudget          = 3 * time.Second
        gapColdBudget            = 3 * time.Second
        gapScanPassTimeout       = 10 * time.Second
        gapScanInterval          = 15 * time.Minute
)

var errGapScanDeferred = errors.New("gap scan deferred while database queue is busy")

func (cdb *PgDB) storedGaps(ctx context.Context, stream int32, start, end int64) (gaps []gap, err error) {
        queryCtx, cancel := context.WithTimeout(ctx, gapQueryTimeout)
        defer cancel()
        rows, err := cdb.Query(queryCtx, cdb.stmtSelectGapWindow, stream, start, end)
        if err != nil {
                return nil, err
        }
        expected := start
        for rows.Next() {
                var index int64
                if err = rows.Scan(&index); err != nil {
                        break
                }
                if index > expected {
                        gaps = append(gaps, gap{start: expected, end: index - 1})
                }
                expected = index + 1
        }
        if err == nil {
                err = rows.Err()
        }
        rows.Close()
        if err == nil && expected <= end {
                gaps = append(gaps, gap{start: expected, end: end})
        }
        if err != nil {
                gaps = nil
        }
        return
}

func (cdb *PgDB) sendGapWindow(ctx context.Context, ls *LogStream, gapCh chan gap, start, end int64, cursor *int64) (found bool, err error) {
        if cdb.QueueUsage() > 50 {
                return false, errGapScanDeferred
        }
        var gaps []gap
        if gaps, err = cdb.storedGaps(ctx, ls.Id, start, end); err != nil {
                return
        }
        verified := end
        if len(gaps) > 0 {
                verified = gaps[0].start
        }
        if verified > *cursor {
                if err = cdb.backfillSetGapStartIndex(ctx, ls, verified); err != nil {
                        return
                }
                *cursor = verified
        }
        for _, g := range gaps {
                found = true
                select {
                case <-ctx.Done():
                        err = ctx.Err()
                        return
                case gapCh <- g:
                        cdb.recordGap(ls, g)
                }
        }
        return
}

func (cdb *PgDB) selectStreamGaps(ctx context.Context, ls *LogStream) (scanErr error) {
        gapCh := ls.getGapCh()
        if gapCh == nil {
                return nil
        }
        defer func() {
                ls.mu.Lock()
                if ls.gapCh == gapCh {
                        close(gapCh)
                }
                ls.mu.Unlock()
                if scanErr != nil && !errors.Is(scanErr, context.Canceled) && !errors.Is(scanErr, errGapScanDeferred) {
                        _ = cdb.LogError(scanErr, "selectStreamGaps", "stream", ls.Id, "url", ls.URL())
                }
        }()
        if cdb.QueueUsage() > 50 {
                return errGapScanDeferred
        }
        select {
        case cdb.gapScanPermit <- struct{}{}:
                defer func() { <-cdb.gapScanPermit }()
        case <-ctx.Done():
                return ctx.Err()
        }
        ctx, cancelPass := context.WithTimeout(ctx, gapScanPassTimeout)
        defer cancelPass()
        if cdb.QueueUsage() > 50 {
                return errGapScanDeferred
        }
        var maxRow sql.NullInt64
        if scanErr = cdb.QueryRow(ctx, cdb.stmtSelectMaxIdx, ls.Id).Scan(&maxRow); scanErr != nil || !maxRow.Valid {
                return
        }
        maxIndex := maxRow.Int64
        var cursor int64
        if scanErr = cdb.QueryRow(ctx, cdb.stmtSelectBackfillIdx, ls.Id).Scan(&cursor); scanErr != nil {
                return
        }
        recentCutoff := max(int64(0), maxIndex-gapRecentLookback+1)
        recentStart := recentCutoff
        var first sql.NullInt64
        if scanErr = cdb.QueryRow(ctx, cdb.stmtSelectMinIdx, ls.Id).Scan(&first); scanErr != nil {
                return
        }
        if first.Valid && first.Int64 > recentStart {
                recentStart = first.Int64
        }
        recentCtx, cancel := context.WithTimeout(ctx, gapRecentBudget)
        hotFloor := maxIndex + 1
        firstGap := int64(-1)
        for end := maxIndex; end >= recentStart && scanErr == nil; {
                start := max(recentStart, end-gapWindowSize+1)
                var gaps []gap
                if cdb.QueueUsage() > 50 {
                        scanErr = errGapScanDeferred
                        break
                }
                gaps, scanErr = cdb.storedGaps(recentCtx, ls.Id, start, end)
                if scanErr != nil {
                        break
                }
                for _, g := range gaps {
                        if firstGap < 0 || g.start < firstGap {
                                firstGap = g.start
                        }
                        select {
                        case <-recentCtx.Done():
                                scanErr = recentCtx.Err()
                        case gapCh <- g:
                                cdb.recordGap(ls, g)
                        }
                        if scanErr != nil {
                                break
                        }
                }
                if scanErr != nil {
                        break
                }
                hotFloor = start
                if start == 0 {
                        break
                }
                end = start - 1
        }
        hotErr := scanErr
        hotBudgetExpired := errors.Is(recentCtx.Err(), context.DeadlineExceeded)
        cancel()
        if hotErr != nil && (ctx.Err() != nil || !errors.Is(hotErr, context.DeadlineExceeded)) {
                return
        }
        scanErr = nil
        if !hotBudgetExpired {
                defer func() {
                        if scanErr == nil {
                                scanErr = hotErr
                        }
                }()
        }
        if firstGap >= 0 && firstGap < cursor {
                if _, scanErr = cdb.Exec(ctx, cdb.stmtLowerBackfillIdx, firstGap, ls.Id); scanErr != nil {
                        return
                }
                cursor = firstGap
        }
        if cursor == 0 {
                if first.Valid && first.Int64 > 0 {
                        // Reverse backfill owns the prefix before the first stored entry.
                        if scanErr = cdb.backfillSetGapStartIndex(ctx, ls, first.Int64); scanErr != nil {
                                return
                        }
                        cursor = first.Int64
                }
        }
        if recentCutoff == 0 && hotFloor == recentStart && cursor >= recentStart {
                verified := maxIndex
                if firstGap >= 0 {
                        verified = firstGap
                }
                if verified > cursor {
                        if scanErr = cdb.backfillSetGapStartIndex(ctx, ls, verified); scanErr != nil {
                                return
                        }
                }
                return
        }
        coldEnd := hotFloor - 1
        coldCtx, cancel := context.WithTimeout(ctx, gapColdBudget)
        defer cancel()
        for cursor <= coldEnd && scanErr == nil {
                if cdb.QueueUsage() > 50 {
                        return errGapScanDeferred
                }
                end := min(coldEnd, cursor+gapWindowSize-1)
                var found bool
                found, scanErr = cdb.sendGapWindow(coldCtx, ls, gapCh, cursor, end, &cursor)
                if found || end == coldEnd {
                        break
                }
                cursor = end + 1
        }
        return
}

func (rlt *requestLogTransport) logRequest(req *http.Request, resp *http.Response, err error) {
        if rlt != nil && rlt.writer != nil {
                if !errors.Is(err, context.Canceled) {
                        var scheme, host, method, uri, result string
                        reqcl := int64(-1)
                        respcl := int64(-1)
                        if req != nil {
                                reqcl = req.ContentLength
                                host = req.Host
                                method = req.Method
                                if req.URL != nil {
                                        scheme = req.URL.Scheme
                                        uri = req.URL.Path
                                        if raw := req.URL.RawQuery; raw != "" {
                                                uri += "?" + raw
                                        }
                                }
                                if resp != nil {
                                        if result = resp.Status; result == "" {
                                                result = fmt.Sprintf("%03d", resp.StatusCode)
                                        }
                                        respcl = resp.ContentLength
                                } else {
                                        result = "000 missing response"
                                }
                        } else {
                                result = "000 missing request"
                        }
                        if err != nil {
                                result += "; " + err.Error()
                        }
                        rlt.mu.Lock()
                        _, _ = fmt.Fprintf(rlt.writer, "%s %s %s://%s%s (%d) => %q (%d)\n",
                                time.Now().UTC().Format(time.RFC3339), method, scheme, host, uri, reqcl, result, respcl)
                        rlt.mu.Unlock()
                }
        }
}

// retryAfterTransport shares a cooldown across one stream's matching requests.
//
// Head and tail clients have separate cooldowns. Classic endpoints have
// separate cooldowns, while a tiled log's tile paths share one. The client
// using this transport has no Client.Timeout; each network request gets the
// original timeout after the cooldown wait so a long Retry-After is usable.
type retryAfterTransport struct {
        next    http.RoundTripper
        timeout time.Duration
        mu      sync.Mutex
        until   map[string]time.Time
}

type cancelBody struct {
        io.ReadCloser
        cancel context.CancelFunc
}

func (body *cancelBody) Read(p []byte) (n int, err error) {
        n, err = body.ReadCloser.Read(p)
        if err != nil {
                body.cancel()
        }
        return
}

func (body *cancelBody) Close() (err error) {
        err = body.ReadCloser.Close()
        body.cancel()
        return
}

func newProviderClient(base *http.Client) *http.Client {
        next := base.Transport
        if next == nil {
                next = http.DefaultTransport
        }
        return &http.Client{
                Transport:     &retryAfterTransport{next: next, timeout: base.Timeout, until: make(map[string]time.Time)},
                CheckRedirect: base.CheckRedirect,
                Jar:           base.Jar,
        }
}

func retryAfterKey(u *url.URL) string {
        path := u.Path
        if idx := strings.LastIndex(path, "/tile/"); idx >= 0 {
                path = path[:idx+len("/tile/")]
        }
        return u.Scheme + "://" + u.Host + path
}

func retryAfterTime(value string, now time.Time) (until time.Time) {
        if seconds, err := strconv.ParseInt(value, 10, 64); err == nil {
                if seconds >= 0 && seconds <= int64((1<<63-1)/int64(time.Second)) {
                        until = now.Add(time.Duration(seconds) * time.Second)
                }
        } else {
                if until, err = http.ParseTime(value); err != nil {
                        // Skylight sends RFC1123 dates with UTC instead of HTTP's GMT.
                        until, _ = time.Parse(time.RFC1123, value)
                }
        }
        if !until.After(now) {
                until = time.Time{}
        }
        return
}

func (transport *retryAfterTransport) wait(ctx context.Context, key string) (err error) {
        for err == nil {
                transport.mu.Lock()
                until := transport.until[key]
                transport.mu.Unlock()
                delay := time.Until(until)
                if delay <= 0 {
                        return ctx.Err()
                }
                err = sleep(ctx, delay)
        }
        return
}

func (transport *retryAfterTransport) RoundTrip(req *http.Request) (resp *http.Response, err error) {
        key := retryAfterKey(req.URL)
        if err = transport.wait(req.Context(), key); err != nil {
                return
        }

        request := req
        var cancel context.CancelFunc
        if transport.timeout > 0 {
                var ctx context.Context
                ctx, cancel = context.WithTimeout(req.Context(), transport.timeout)
                request = req.Clone(ctx)
        }
        resp, err = transport.next.RoundTrip(request)
        if err != nil || resp == nil {
                if cancel != nil {
                        cancel()
                }
                return
        }
        if resp.StatusCode == http.StatusTooManyRequests || resp.StatusCode == http.StatusServiceUnavailable {
                if until := retryAfterTime(resp.Header.Get("Retry-After"), time.Now()); !until.IsZero() {
                        // Keep the operator's full deadline; cancellation still ends the wait.
                        transport.mu.Lock()
                        if until.After(transport.until[key]) {
                                transport.until[key] = until
                        }
                        transport.mu.Unlock()
                }
        }
        if cancel != nil {
                if resp.Body != nil {
                        resp.Body = &cancelBody{ReadCloser: resp.Body, cancel: cancel}
                } else {
                        cancel()

func getCacheDir(baseDir string, monitoringURL string) (cacheDir string) {
        if baseDir != "" {
                cacheDir = path.Join(baseDir, "tilecache")
                if monitoringURL != "" {
                        cacheDir = path.Join(cacheDir, urlToFileString(monitoringURL))
                }
        }
        return
}

func pruneCacheFiles(cacheDir string, maxAge time.Duration, now time.Time) (removed int, err error) {
        cutoff := now.Add(-maxAge)
        var dirs []string
        var walkErr error
        walkErr = filepath.WalkDir(cacheDir, func(path string, d fs.DirEntry, walkErr error) error {
                err = errors.Join(err, walkErr)
                if walkErr == nil {
                        if d.IsDir() {
                                if path != cacheDir {
                                        dirs = append(dirs, path)
                                }
                        } else {
                                var info fs.FileInfo
                                info, walkErr = d.Info()
                                err = errors.Join(err, walkErr)
                                if walkErr == nil {
                                        if info.ModTime().Before(cutoff) {
                                                walkErr = os.Remove(path)
                                                err = errors.Join(err, walkErr)
                                                if walkErr == nil {
                                                        removed++
                                                }
                                        }
                                }
                        }
                }
                return nil
        })

        err = errors.Join(err, walkErr)

        // try removing directories
        for i := len(dirs) - 1; i >= 0; i-- {
                if walkErr = os.Remove(dirs[i]); walkErr != nil {
                        if !errors.Is(walkErr, syscall.ENOTEMPTY) && !errors.Is(walkErr, os.ErrNotExist) {
                                err = errors.Join(err, walkErr)
                        }
                }
        }
        return
}

func (cs *CertStream) runCachePruner(ctx context.Context, wg *sync.WaitGroup, cacheDir string, maxAge time.Duration) {
        ticker := time.NewTicker(min(time.Minute, maxAge))
        defer func() {
                wg.Done()
                ticker.Stop()
        }()
        for {
                if _, err := pruneCacheFiles(cacheDir, maxAge, time.Now()); err != nil {
                        _ = cs.LogError(err, "runCachePruner", "dir", cacheDir)
                }
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                }
        }
}

        var file *os.File
        if file, err = os.OpenFile(lw.path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, lw.perm); err == nil {
                n, err = file.Write(p)

func (cs *CertStream) ensureOperator(ctx context.Context, op *loglist3.Operator, opDom string) (logop *LogOperator, err error) {
        cs.mu.Lock()
        logop = cs.operators[opDom]
        cs.mu.Unlock()

        if logop == nil {
                logop = &LogOperator{
                        CertStream: cs,
                        operator:   op,
                        Domain:     opDom,
                        streams:    make(map[string]*LogStream),
                }
                sort.Strings(op.Email)
                if err = cs.DB().ensureOperator(ctx, logop); err == nil {
                        cs.mu.Lock()
                        cs.operators[opDom] = logop
                        cs.mu.Unlock()
                }
        }
        return
}

func (cs *CertStream) ensureOperatorAndLog(ctx context.Context, op *loglist3.Operator, log *loglist3.Log, wg *sync.WaitGroup) (err error) {
        cs.lifecycleMu.Lock()
        defer cs.lifecycleMu.Unlock()
        opDom := OperatorDomain(log.URL)

        var logop *LogOperator
        if logop, err = cs.ensureOperator(ctx, op, opDom); err == nil {
                err = logop.ensureStream(ctx, log, wg)
        }

        return
}

func (cs *CertStream) ensureOperatorAndTiledLog(ctx context.Context, op *loglist3.Operator, log *loglist3.TiledLog, wg *sync.WaitGroup) (err error) {
        cs.lifecycleMu.Lock()
        defer cs.lifecycleMu.Unlock()
        opDom := OperatorDomain(log.MonitoringURL)

        var logop *LogOperator
        if logop, err = cs.ensureOperator(ctx, op, opDom); err == nil {
                err = logop.ensureTiledStream(ctx, log, wg)
        }

        return
}

func (cs *CertStream) updateStreams(ctx context.Context, wg *sync.WaitGroup) (err error) {
        defer wg.Done()
        if ctx.Err() == nil {
                var logList *loglist3.LogList
                if logList, err = getLogList(ctx, cs.HeadClient, loglist3.AllLogListURL); err == nil {
                        states := make(map[string]*loglist3.LogStates)
                        addState := func(url string, state *loglist3.LogStates) bool {
                                status := state.LogStatus()
                                if status == loglist3.UndefinedLogStatus {
                                        return false
                                }
                                if status == loglist3.RetiredLogStatus {
                                        if prior, ok := states[url]; ok && prior.LogStatus() != loglist3.RetiredLogStatus {
                                                return false
                                        }
                                }
                                states[url] = state
                                return status == loglist3.UsableLogStatus
                        }
                        for _, op := range logList.Operators {
                                for _, log := range op.Logs {
                                        if addState(log.URL, log.State) {
                                                if e := cs.ensureOperatorAndLog(ctx, op, log, wg); e != nil {
                                                        err = errors.Join(err, e)
                                                }
                                        }
                                }
                                for _, log := range op.TiledLogs {
                                        if addState(log.MonitoringURL, log.State) {
                                                if e := cs.ensureOperatorAndTiledLog(ctx, op, log, wg); e != nil {
                                                        err = errors.Join(err, e)
                                                }
                                        }
                                }
                        }
                        if ctx.Err() == nil {
                                retired := make(map[string]bool)
                                for url, state := range states {
                                        if state.LogStatus() == loglist3.RetiredLogStatus {
                                                retired[url] = true
                                        }
                                }
                                _ = cs.LogError(cs.retireStreams(ctx, retired, states), "retireStreams")
                        }
                }

                var operators []string
                for _, lo := range cs.Operators() {
                        operators = append(operators, fmt.Sprintf("%s*%d", lo.Domain, len(lo.Streams())))
                }
                slices.Sort(operators)
                cs.LogInfo("active", "streams", operators)
        }
        return
}

func (cs *CertStream) removeStream(ls *LogStream) {
        cs.lifecycleMu.Lock()
        defer cs.lifecycleMu.Unlock()
        lo := ls.LogOperator
        lo.mu.Lock()
        if lo.streams[ls.URL()] == ls {
                delete(lo.streams, ls.URL())
        }
        empty := len(lo.streams) == 0
        lo.mu.Unlock()
        if empty {
                cs.mu.Lock()
                if cs.operators[lo.Domain] == lo {
                        delete(cs.operators, lo.Domain)
                }
                cs.mu.Unlock()
        }
}

func (cs *CertStream) retireStreams(ctx context.Context, retired map[string]bool, states map[string]*loglist3.LogStates) (err error) {
        db := cs.DB()
        var ids []int32
        if db != nil {
                urls := make([]string, 0, len(states))
                values := make([]string, 0, len(states))
                for url, state := range states {
                        var value []byte
                        if value, err = json.Marshal(state); err != nil {
                                break
                        }
                        urls = append(urls, url)
                        values = append(values, string(value))
                }
                if err == nil && len(urls) > 0 {
                        _, err = db.Exec(ctx, db.Pfx(`UPDATE CERTDB_stream AS s
SET json = jsonb_set(s.json::jsonb, '{state}', v.state::jsonb)::text
FROM unnest($1::text[], $2::text[]) AS v(url, state)
WHERE s.url = v.url
  AND COALESCE((s.json::jsonb->'state') ? 'retired', FALSE) <> (v.state::jsonb ? 'retired')`), urls, values)
                }
                if err == nil {
                        rows, e := db.Query(ctx, db.Pfx(`SELECT id, url FROM CERTDB_stream WHERE (json::jsonb->'state') ? 'retired' ORDER BY id`))
                        err = e
                        if err == nil {
                                for rows.Next() {
                                        var id int32
                                        var url string
                                        if err = rows.Scan(&id, &url); err != nil {
                                                break
                                        }
                                        ids = append(ids, id)
                                        retired[url] = true
                                }
                                if err == nil {
                                        err = rows.Err()
                                }
                                rows.Close()
                        }
                }
        }
        var stopped []*LogStream
        for _, op := range cs.Operators() {
                for _, ls := range op.Streams() {
                        if retired[ls.URL()] && ls.cancel != nil {
                                ls.cancel()
                                stopped = append(stopped, ls)
                        }
                }
        }
        for _, ls := range stopped {
                select {
                case <-ctx.Done():
                        return ctx.Err()
                case <-ls.done:
                }
        }
        if err != nil || db == nil || cs.OnRetiredStreams == nil || len(ids) == 0 {
                return
        }
        if err = db.flushWorkers(ctx); err != nil {
                return
        }
        err = cs.OnRetiredStreams(ctx, db, ids)
        return
}
